project(CalicataModelChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.9.3 REQUIRED COMPONENTS Core Gui Qml)

target_include_directories(calicata_model_checks PRIVATE ../..)
target_compile_definitions(calicata_model_checks PRIVATE INGE_MOBILE)
target_link_libraries(calicata_model_checks PRIVATE Qt6::Core Qt6::Gui Qt6::Qml)
