cmake_minimum_required(VERSION 3.21)
project(RetroplasmaHostTests LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.9.3 REQUIRED COMPONENTS Core Gui Quick Concurrent ShaderTools)
enable_testing()
add_subdirectory("../../src/retroplasma" retroplasma)
add_executable(rp_core_tests core_tests.cpp)
target_link_libraries(rp_core_tests PRIVATE InGeRetroplasma)
add_test(NAME rp_core COMMAND rp_core_tests)
add_executable(rp_provider_tests provider_tests.cpp)
target_link_libraries(rp_provider_tests PRIVATE InGeRetroplasma)
target_compile_definitions(rp_provider_tests PRIVATE RP_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../src/retroplasma/resources/sample")
add_test(NAME rp_provider COMMAND rp_provider_tests)
add_executable(rp_planet_tests planet_tests.cpp)
target_link_libraries(rp_planet_tests PRIVATE InGeRetroplasma)
add_test(NAME rp_planet COMMAND rp_planet_tests)
add_executable(rp_planet_provider_tests planet_provider_tests.cpp)
target_link_libraries(rp_planet_provider_tests PRIVATE InGeRetroplasma)
add_test(NAME rp_planet_provider COMMAND rp_planet_provider_tests)
add_executable(rp_planet_view_tests planet_view_tests.cpp)
target_link_libraries(rp_planet_view_tests PRIVATE InGeRetroplasma InGeRetroplasmaplugin)
target_compile_definitions(rp_planet_view_tests PRIVATE RP_VIEW_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/../../qml/Mobile/RetroplasmaEarthView.qml")
add_test(NAME rp_planet_view COMMAND rp_planet_view_tests)
set_tests_properties(rp_planet_view PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_CONTROLS_STYLE=Basic")
